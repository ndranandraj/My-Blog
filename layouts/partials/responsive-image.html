{{- /* responsive-image.html — emit a with WebP + original-format srcset for an image that lives under assets/ (so Hugo Pipes can process it). Why this exists: every published image is referenced by an absolute path like "/images/foo.jpg". Files under static are copied verbatim and CANNOT be processed by Hugo, so the old render hooks always fell through to a full-size . We now keep a processing master under assets/images/ with the same filename; this partial resolves it via resources.Get, generates resized WebP + original-format variants, and emits a . Safe-by-fallback: it emits a plain lazy at the original src — the previous behaviour — when there is no asset master (SVGs, external URLs), or when the asset cannot be decoded. That last case matters: some files carry a misleading extension (e.g. WebP bytes saved as .jpg). Hugo trusts the extension and then fails to decode, so we probe the resize with `try` and fall back cleanly instead of failing the whole build. Browsers sniff the real content, so the fallback still renders. Params (dict): src (required) e.g. "/images/foo.jpg" alt alt text (default "") title optional title attribute class optional class attribute sizes default "(min-width: 768px) 720px, 100vw" loading "lazy" (default) or "eager" for above-the-fold / LCP images decoding "async" (default) fetchpriority optional, e.g. "high" for the LCP cover */ -}} {{- $src := .src -}} {{- $alt := .alt | default "" -}} {{- $title := .title -}} {{- $class := .class -}} {{- $sizes := .sizes | default "(min-width: 768px) 720px, 100vw" -}} {{- $loading := .loading | default "lazy" -}} {{- $decoding := .decoding | default "async" -}} {{- $fetchpriority := .fetchpriority -}} {{- $widths := slice 480 720 1080 1500 -}} {{- $img := "" -}} {{- if and $src (hasPrefix $src "/") -}} {{- $img = resources.Get (strings.TrimPrefix "/" $src) -}} {{- end -}} {{- /* Decide whether we can actually process this image. Guard the extension check with a real resize probe so a mislabeled/undecodable file falls back instead of erroring the build. */ -}} {{- $canProcess := false -}} {{- if and $img (in (slice "jpeg" "png") $img.MediaType.SubType) -}} {{- $probe := try ($img.Resize "480x webp q80") -}} {{- if not $probe.Err -}}{{- $canProcess = true -}}{{- end -}} {{- end -}} {{- if $canProcess -}} {{- $webpset := slice -}} {{- $origset := slice -}} {{- range $w := $widths -}} {{- if ge $img.Width $w -}} {{- $orig := $img.Resize (printf "%dx q82" $w) -}} {{- $webp := $img.Resize (printf "%dx webp q80" $w) -}} {{- $origset = $origset | append (printf "%s %dw" $orig.RelPermalink $w) -}} {{- $webpset = $webpset | append (printf "%s %dw" $webp.RelPermalink $w) -}} {{- end -}} {{- end -}} {{- /* Always include a candidate at the native width so large viewports get full quality. */ -}} {{- $webpfull := $img.Resize (printf "%dx webp q80" $img.Width) -}} {{- $origset = $origset | append (printf "%s %dw" $img.RelPermalink $img.Width) -}} {{- $webpset = $webpset | append (printf "%s %dw" $webpfull.RelPermalink $img.Width) -}} {{- else -}} {{- end -}}
