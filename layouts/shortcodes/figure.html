{{- /* Custom figure shortcode override. Adds: loading="lazy", decoding="async", async style. If the image lives in the page bundle, we also emit width/height + a responsive srcset. Falls back to a plain for absolute/static URLs so nothing breaks. */ -}} {{- $src := .Get "src" -}} {{- $alt := .Get "alt" -}} {{- $caption := .Get "caption" -}} {{- $title := .Get "title" -}} {{- $class := .Get "class" -}} {{- $link := .Get "link" -}} {{- $target := .Get "target" | default "_blank" -}} {{- $rel := .Get "rel" | default "noopener noreferrer" -}} {{- $img := "" -}} {{- if $src -}} {{- if not (or (hasPrefix $src "http") (hasPrefix $src "/")) -}} {{- $img = .Page.Resources.GetMatch $src -}} {{- end -}} {{- end -}} {{- if $link }}{{ end -}} {{- if $img -}} {{- $sizes := slice 480 720 1080 1500 -}} {{- $srcset := slice -}} {{- range $sizes -}} {{- if ge $img.Width . -}} {{- $resized := $img.Resize (printf "%dx q82" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $resized.RelPermalink .) -}} {{- end -}} {{- end -}} {{- $srcset = $srcset | append (printf "%s %dw" $img.RelPermalink $img.Width) -}} {{- else -}} {{- end -}} {{- if $link }}{{ end -}} {{- with $caption -}} {{- end -}}
