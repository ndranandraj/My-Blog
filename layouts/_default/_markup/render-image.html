{{- /* Markdown image render hook. Three paths, in priority order: 1. Absolute "/images/..." src → responsive-image partial (resolves an assets/images/ master, emits WebP + srcset, falls back to plain ). 2. Page-bundle relative src → resize + srcset from the page resource. 3. Anything else (external) → plain lazy . */ -}} {{- $src := .Destination -}} {{- $alt := .Text -}} {{- $title := .Title -}} {{- if hasPrefix $src "/images/" -}} {{- partial "responsive-image.html" (dict "src" $src "alt" $alt "title" $title) -}} {{- else -}} {{- $img := "" -}} {{- if not (or (hasPrefix $src "http") (hasPrefix $src "/")) -}} {{- $img = .Page.Resources.GetMatch $src -}} {{- end -}} {{- if $img -}} {{- $sizes := slice 480 720 1080 1500 -}} {{- $srcset := slice -}} {{- range $sizes -}} {{- if ge $img.Width . -}} {{- $resized := $img.Resize (printf "%dx q82" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $resized.RelPermalink .) -}} {{- end -}} {{- end -}} {{- $srcset = $srcset | append (printf "%s %dw" $img.RelPermalink $img.Width) -}} {{- else -}} {{- end -}} {{- end -}}
